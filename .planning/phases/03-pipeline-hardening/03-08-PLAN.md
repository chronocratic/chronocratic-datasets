    ```python
    def _load_clean_representations(
        *,
        run_dir: Path,
        task_name: str,
    ) -> TaskRepresentationBundle:
        """Load a TaskRepresentationBundle from persisted NPZ files (resume fallback).

        Reads the same .npz files written by _persist_clean_representations.
        Raises FileNotFoundError if any required file is missing.
        """
        base = run_dir / 'representations' / task_name / 'clean'
        partitions = {}
        for split in ('train', 'valid', 'test'):
            npz_path = base / f'{split}.npz'
            if not npz_path.exists():
                msg = f'Missing persisted clean rep: {npz_path}'
                raise FileNotFoundError(msg)
            loaded = np.load(file=str(npz_path))
            partitions[split] = PartitionRepresentations(
                features=loaded['features'],
                labels=loaded['labels'],
            )
        return TaskRepresentationBundle(
            train=partitions['train'],
            valid=partitions['valid'],
            test=partitions['test'],
        )
    ```

    ```python
    def _load_shared_attacked_inputs_from_disk(
        *,
        run_dir: Path,
        attacks: tuple[AttackRunConfig, ...],
        first_task_name: str,
    ) -> dict[str, tuple[Tensor, dict[str, Any]]]:
        """Reconstruct shared_attacked_inputs from persisted attacked reps.

        Loads the test features from each attack's persisted NPZ. These are encoded
        representations, not raw inputs — sufficient for downstream evaluation.
        Raises FileNotFoundError if persisted files are missing.
        """
        result: dict[str, tuple[Tensor, dict[str, Any]]] = {}
        for attack_config in attacks:
            attack_name = attack_config.name
            attack_dir = run_dir / 'representations' / first_task_name / 'attacks' / attack_name
            npz_path = attack_dir / 'test.npz'
            if not npz_path.exists():
                msg = f'Missing persisted attacked inputs: {npz_path}'
                raise FileNotFoundError(msg)
            loaded = np.load(file=str(npz_path))
            features = torch.as_tensor(loaded['features'])
            meta_path = attack_dir / 'attack_metadata.json'
            metadata = {}
            if meta_path.exists():
                with meta_path.open(mode='r', encoding='utf-8') as fh:
                    metadata = json.load(fh)
            result[attack_name] = (features, metadata)
        return result
    ```
