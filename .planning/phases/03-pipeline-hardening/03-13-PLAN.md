    """Unit tests for pipeline state management."""

    from pathlib import Path

        STATE_FILENAME,
        PipelineState,
        _PipelineStateBuilder,

        load_pipeline_state,

    def test_state_global_completion() -> None:
        """Marking a step without task_name marks global completion."""
        builder = _PipelineStateBuilder(config_hash='abc12345')
        builder.mark_complete(step='train')
        state = builder.build()
        assert state.is_step_complete(step='train')
        assert not state.is_step_complete(step='encoding')


    def test_state_task_level_completion() -> None:
        """Per-task granularity: only marked tasks are complete."""
        builder = _PipelineStateBuilder(config_hash='abc12345')
        builder.mark_complete(step='encoding', task_name='classification')
        state = builder.build()
        assert state.is_step_complete(step='encoding', task_name='classification')
        assert not state.is_step_complete(step='encoding', task_name='forecasting')


    def test_state_no_duplicate_task() -> None:
        """Marking the same step+task twice does not duplicate."""
        builder = _PipelineStateBuilder(config_hash='abc12345')
        builder.mark_complete(step='encoding', task_name='classification')
        builder.mark_complete(step='encoding', task_name='classification')
        state = builder.build()
        assert state.completed['encoding'] == ['classification']


    def test_state_incomplete_step() -> None:
        """Unmarked step returns False."""
        state = PipelineState(
            completed={'train': []},
            config_hash='abc',

            last_updated='2026-01-01T00:00:00+00:00',
        )
        assert not state.is_step_complete(step='encoding')

        """to_dict -> from_dict produces equivalent state."""

            last_updated='2026-01-01T01:00:00+00:00',
        )
        restored = from_dict(data=to_dict(state=original))
        assert restored.completed == original.completed
        assert restored.config_hash == original.config_hash
        assert restored.started_at == original.started_at
        assert restored.last_updated == original.last_updated

    from src.rbspaper.pipeline.state import (
        STATE_FILENAME,
        _atomic_write_json,
        PipelineState,
        ...
    )

    # === Atomic write tests ===

    def test_atomic_write_creates_file(tmp_path: Path) -> None:
        """_atomic_write_json creates target file, no .tmp left behind."""
        target = tmp_path / 'test.json'
        _atomic_write_json(path=target, data={'key': 'value'})
        assert target.exists()
        assert not (tmp_path / 'test.json.tmp').exists()


    def test_save_load_round_trip(tmp_path: Path) -> None:
        """save_pipeline_state -> load_pipeline_state preserves data."""
        state = _PipelineStateBuilder(config_hash='test1234').build()
        builder = _PipelineStateBuilder(config_hash='test1234')
        builder.mark_complete(step='train')
        state = builder.build()
        path = tmp_path / STATE_FILENAME
        save_pipeline_state(state=state, path=path)
        loaded = load_pipeline_state(path=path)
        assert loaded.is_step_complete(step='train')
        assert loaded.config_hash == 'test1234'


    def test_load_missing_file(tmp_path: Path) -> None:
        """load_pipeline_state raises FileNotFoundError for missing file."""
        try:
            load_pipeline_state(path=tmp_path / 'nonexistent.json')
            raise AssertionError('Expected FileNotFoundError')
        except FileNotFoundError:
            pass


    # === Hash tests ===

    def test_hash_deterministic() -> None:
        """Same params + seed produce identical hash."""
        params = {'learning_rate': 1e-3, 'depth': 10}
        h1 = compute_config_hash(model_params=params, seed=42)
        h2 = compute_config_hash(model_params=params, seed=42)
        assert h1 == h2


    def test_hash_differs_on_param_change() -> None:
        """Different params produce different hash."""
        h1 = compute_config_hash(model_params={'lr': 0.001}, seed=42)
        h2 = compute_config_hash(model_params={'lr': 0.01}, seed=42)
        assert h1 != h2

        """Hash is exactly 8 hex characters."""
        h = compute_config_hash(model_params={'a': 1}, seed=1)
        assert len(h) == 8
        int(h, 16)  # Validates hex format
