def get_dataset_metadata(*, dataset_name: str, registry: tuple[DatasetMetadata, ...] = DATASET_REGISTRY) -> DatasetMetadata:
    for dataset in registry:
        if dataset.name == dataset_name:
            return dataset
    raise KeyError(f'unknown dataset: {dataset_name}')

def list_datasets_for_families(*, families: tuple[str, ...], registry: tuple[DatasetMetadata, ...] = DATASET_REGISTRY) -> list[DatasetMetadata]:
    allowed_families = set(families)
    return [dataset for dataset in registry if dataset.family in allowed_families]
