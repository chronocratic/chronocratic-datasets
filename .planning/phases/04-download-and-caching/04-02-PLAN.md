```python
def download_ucr_uea(*, config: ClassificationConfig, overwrite_cache: bool = False) -> dict[str, Path]:
    cache_root = get_cache_dir()
    dataset_dir = cache_root / config.name
    archive_name = f"{config.name}.zip"
    archive_path = download_file(url=str(config.url), sha256=config.sha256, cache_dir=dataset_dir, filename=archive_name, overwrite_cache=overwrite_cache)
    extract_dir = dataset_dir / "extracted"
    extract_dir.mkdir(exist_ok=True)
    with zipfile.ZipFile(archive_path) as zf:
        zf.extractall(extract_dir)
    return {"train": ..., "test": ...}
```

def download_forecasting(*, config: ForecastingConfig, overwrite_cache: bool = False) -> Path:
