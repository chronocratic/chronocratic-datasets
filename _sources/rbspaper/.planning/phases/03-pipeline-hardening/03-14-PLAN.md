    ```python
    def test_resume_skips_completed_steps(
        tmp_path: Path, monkeypatch: pytest.MonkeyPatch
    ) -> None:
        """Pipeline skips steps marked complete in previous_state."""
        from src.rbspaper.pipeline.core import (
            _train_model as original_train,
        )

        train_called = False

        def _counting_train(**kwargs: object) -> object:
            nonlocal train_called
            train_called = True
            return original_train(**kwargs)

        monkeypatch.setattr('src.rbspaper.pipeline.core.encode_data', _fake_encode_for_resume)
        monkeypatch.setattr('src.rbspaper.pipeline.core.execute_attack', _fake_attack_for_resume)
        monkeypatch.setattr('src.rbspaper.pipeline.core.evaluate', _fake_eval_for_resume)
        monkeypatch.setattr('src.rbspaper.pipeline.core._train_model', _counting_train)

        # Build state with train marked complete
        builder = _PipelineStateBuilder(config_hash='abc12345')
        builder.mark_complete(step='train')
        previous = builder.build()

        config = _build_config(tmp_path, run_name='resume_test')

        run_experiment_pipeline(
            config=config,
            previous_state=previous,
            force=False,
        )

        # train should have been skipped
        assert not train_called


    def test_force_overrides_resume(
        tmp_path: Path, monkeypatch: pytest.MonkeyPatch
    ) -> None:
        """force=True ignores previous_state and runs all steps."""
        # This test verifies force=True does not raise and completes
        monkeypatch.setattr('src.rbspaper.pipeline.core.encode_data', _fake_encode_for_resume)
        monkeypatch.setattr('src.rbspaper.pipeline.core.execute_attack', _fake_attack_for_resume)
        monkeypatch.setattr('src.rbspaper.pipeline.core.evaluate', _fake_eval_for_resume)

        builder = _PipelineStateBuilder(config_hash='abc12345')
        builder.mark_complete(step='train')
        previous = builder.build()

        config = _build_config(tmp_path, run_name='force_test')

        # Should not raise, runs all steps despite previous_state
        results = run_experiment_pipeline(
            config=config,
            previous_state=previous,
            force=True,
        )
        assert results is not None
    ```

    **CRITICAL**: The `_TinyModel` class MUST include `model_name = 'TinyModel'` as a class attribute. The pipeline's `ExperimentPipelineResults` constructor uses `getattr(config.model, 'model_name', type(config.model).__name__)`. Without this attribute, the model_name falls back to the class name — which works, but having the attribute explicitly makes tests deterministic and matches the project's model convention.
