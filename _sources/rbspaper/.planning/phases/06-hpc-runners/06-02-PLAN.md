  --- classification/univariate/run_on.sh ---

  Purpose: Run one experiment on one classification/univariate dataset via HPC.
  Family: ucr
  No --forecasting_mode (this is classification).

  Usage: `run_on.sh <experiment_id> <dataset_name> [options]`

  ```bash
  #!/usr/bin/env bash
  #
  # run_on.sh — Run an experiment on a single classification/univariate dataset via HPC.
  #
  # Usage:
  #   ./runners/bash/classification/univariate/run_on.sh <experiment_id> <dataset_name> [options]
  #
  # Options (forwarded to hpc_submit_single.sh):
  #   --seed <N>           Random seed (default: 42)
  #   --num_runs <N>       Number of seed runs (default: 1)
  #   --force              Ignore existing checkpoint state
  #   --dry_run            Generate SLURM script without submitting
  #

  set -uo pipefail

  SCRIPT_DIR=$(cd "$(dirname "${BASH_SOURCE[0]}")" || exit 1; pwd) || exit 1
  PARENT_DIR=$(cd "$SCRIPT_DIR/../../.." || exit 1; pwd) || exit 1

  if [ $# -lt 2 ]; then
    echo "Usage: $0 <experiment_id> <dataset_name> [--seed N] [--num_runs N] [--force] [--dry_run]" >&2
    exit 1
  fi

  experiment_id="$1"
  dataset_name="$2"
  shift 2

  bash "$PARENT_DIR/hpc_submit_single.sh" "$experiment_id" "$dataset_name" --family ucr --task classification "$@"
  ```

  --- classification/univariate/run_all.sh ---

  Purpose: Loop over all classification datasets, submit each via run_on.sh.
  Queries get_classification_datasets() at runtime (no hardcoded dataset list).
  EXPERIMENTS: ts2vec, autotcl (hardcoded per decision).

  ```bash
  #!/usr/bin/env bash
  #
  # run_all.sh — Run all experiments on classification/univariate datasets via HPC.
  #
  # Usage:
  #   ./runners/bash/classification/univariate/run_all.sh [options]
  #
  # Options (forwarded to run_on.sh per experiment):
  #   --seed <N>           Random seed (default: 42)
  #   --num_runs <N>       Number of seed runs (default: 1)
  #   --force              Ignore existing checkpoint state
  #   --dry_run            Generate SLURM script without submitting
  #

  set -uo pipefail

  SCRIPT_DIR=$(cd "$(dirname "${BASH_SOURCE[0]}")" || exit 1; pwd) || exit 1
  PROJECT_ROOT=$(cd "$SCRIPT_DIR/../../../.." || exit 1; pwd) || exit 1
  export PYTHONPATH="${PROJECT_ROOT}:${PYTHONPATH:-}"

  RUN_ON_PATH="${SCRIPT_DIR}/run_on.sh"

  EXPERIMENTS=(
    "ts2vec"
    "autotcl"
  )

  # Query registry for classification dataset list (no hardcoded text files)
  DATASETS=()
  while IFS= read -r line; do
    DATASETS+=("$line")
  done < <(cd "$PROJECT_ROOT" || exit 1; uv run python -c "
import sys
from src.rbspaper.data.data_setup import get_classification_datasets
for name in get_classification_datasets():
    print(name)
  ")

  if [ ${#DATASETS[@]} -eq 0 ]; then
    echo "[ERROR] No classification datasets found in registry" >&2
    exit 1
  fi

  log() {
    echo "[$(date '+%H:%M:%S')] $*"
  }

  for exp_id in "${EXPERIMENTS[@]}"; do
    log "========================================"
    log "Experiment: $exp_id"
    log "========================================"
    for dataset_name in "${DATASETS[@]}"; do
      log "Submitting: $exp_id on $dataset_name"
      bash "$RUN_ON_PATH" "$exp_id" "$dataset_name" "$@"
    done
  done

  log "All classification/univariate experiments submitted."
  ```
